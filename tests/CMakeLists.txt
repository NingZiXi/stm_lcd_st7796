cmake_minimum_required(VERSION 3.22)
project(stm_lcd_st7796_test LANGUAGES C CXX)
enable_testing()
add_compile_options(-Wall -Wextra -Werror -UNDEBUG)
add_subdirectory(.. component)
# 只在测试构建替换 libc 分配符号；生产组件没有分配器选项。
target_compile_definitions(stm_lcd_st7796 PRIVATE calloc=test_calloc free=test_free)
add_executable(test_component test_panel.c test_allocator.c)
target_link_libraries(test_component PRIVATE stm_lcd_st7796)
add_library(header_consumers OBJECT headers.c headers.cpp)
target_compile_features(header_consumers PRIVATE c_std_11 cxx_std_17)
target_link_libraries(header_consumers PRIVATE stm_lcd_st7796)
add_test(NAME contracts COMMAND test_component)
