cmake_minimum_required(VERSION 3.22)
project(stm_lcd_st7796 C)
add_library(stm_lcd_st7796 STATIC src/stm_lcd_st7796.c)
target_include_directories(stm_lcd_st7796 PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include)
target_compile_features(stm_lcd_st7796 PUBLIC c_std_11)

# 优先复用应用 target 或同级源码，独立接入固定 stm_common v1.0.0。
if(NOT TARGET stm_common)
    if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../stm_common/CMakeLists.txt")
        add_subdirectory("${CMAKE_CURRENT_SOURCE_DIR}/../stm_common" "${CMAKE_CURRENT_BINARY_DIR}/stm_common")
    else()
        option(STM_COMMON_FETCH "Download missing stm_common dependency" ON)
        set(STM_COMMON_GIT_REPOSITORY "https://github.com/NingZiXi/stm_common.git" CACHE STRING "stm_common repository or mirror")
        if(NOT STM_COMMON_FETCH)
            message(FATAL_ERROR "stm_common is missing. Provide its target or sibling directory, or enable STM_COMMON_FETCH.")
        endif()
        include(FetchContent)
        FetchContent_Declare(stm_common
            GIT_REPOSITORY "${STM_COMMON_GIT_REPOSITORY}"
            GIT_TAG ce3d186dde2d374a8e9c7b9068a7b88f97d57dc1)
        FetchContent_MakeAvailable(stm_common)
    endif()
endif()
if(NOT TARGET stm_common)
    message(FATAL_ERROR "stm_common dependency did not provide the stm_common target")
endif()
target_link_libraries(stm_lcd_st7796 PUBLIC stm_common)


# 芯片组件共享 stm_lcd 依赖及同一固定提交，不追踪 main。
option(STM_LCD_FETCH "Download missing stm_lcd dependency" ON)
set(STM_LCD_SOURCE_DIR "" CACHE PATH "Explicit local stm_lcd source directory")
set(STM_LCD_GIT_REPOSITORY "https://github.com/NingZiXi/stm_lcd.git" CACHE STRING "stm_lcd repository or trusted mirror")
if(NOT TARGET stm_lcd)
    set(_stm_lcd_local_source "")
    if(STM_LCD_SOURCE_DIR)
        set(_stm_lcd_local_source "${STM_LCD_SOURCE_DIR}")
    elseif(FETCHCONTENT_SOURCE_DIR_STM_LCD)
        set(_stm_lcd_local_source "${FETCHCONTENT_SOURCE_DIR_STM_LCD}")
    elseif(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/../stm_lcd/CMakeLists.txt")
        set(_stm_lcd_local_source "${CMAKE_CURRENT_SOURCE_DIR}/../stm_lcd")
    endif()
    if(_stm_lcd_local_source)
        get_filename_component(_stm_lcd_local_source "${_stm_lcd_local_source}" ABSOLUTE BASE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
        if(NOT EXISTS "${_stm_lcd_local_source}/CMakeLists.txt")
            message(FATAL_ERROR "stm_lcd local source is invalid: ${_stm_lcd_local_source}")
        endif()
        add_subdirectory("${_stm_lcd_local_source}" "${CMAKE_CURRENT_BINARY_DIR}/stm_lcd")
    else()
        if(NOT STM_LCD_FETCH)
            message(FATAL_ERROR "stm_lcd is missing. Provide its target, STM_LCD_SOURCE_DIR, FETCHCONTENT_SOURCE_DIR_STM_LCD or sibling source, or enable STM_LCD_FETCH.")
        endif()
        include(FetchContent)
        if(FETCHCONTENT_FULLY_DISCONNECTED AND NOT EXISTS "${FETCHCONTENT_BASE_DIR}/stm_lcd-src/CMakeLists.txt")
            message(FATAL_ERROR "stm_lcd is missing in fully disconnected mode. Provide its target or local source directory.")
        endif()
        FetchContent_Declare(stm_lcd
            GIT_REPOSITORY "${STM_LCD_GIT_REPOSITORY}"
            GIT_TAG 7e7c3f26d43881f9b14ae4268a3be20404a93835 # stm_lcd v1.1.0
        )
        FetchContent_MakeAvailable(stm_lcd)
    endif()
endif()
if(NOT TARGET stm_lcd)
    message(FATAL_ERROR "stm_lcd dependency did not provide the stm_lcd target")
endif()
target_link_libraries(stm_lcd_st7796 PUBLIC stm_lcd)
